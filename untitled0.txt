PERCENT_PROFIT = 0.12  # نسبة ربح تطبيق مندوبكم 12%
FIXED_SERVICE_FEE = 30  # الرسوم الثابتة كخدمة مع كل المنتجات

def price_it(price, weight=6):
    """
    حساب السعر النهائي للمنتج بناءً على المعادلة المعتمدة الرسمية:
    السعر = سعر ايكيا + (سعر ايكيا * 12%) + (الوزن * 2 ريال) + 30 ريال ثابتة خدمة
    """
    profit_amount = price * PERCENT_PROFIT
    shipping_cost = weight * 2
    total_price = price + profit_amount + shipping_cost + FIXED_SERVICE_FEE
    return ceil(total_price)

def extract_product_details(user_input):
    cleaned_url = user_input.replace(".", "").replace("-", "")
    combo_match = re.search(r's(\d{8})', cleaned_url, re.IGNORECASE)
    if combo_match:
        return combo_match.group(1), "SPR"
    item_match = re.search(r'(\d{8})', cleaned_url)
    if item_match:
        return item_match.group(1), "ART"
    return None, None

#  المسبار الذكي لمخزون فرع جدة (110)
# ==========================================
def extract_stock_from_html(html_content, prod_id):
    """ استخراج المخزون الخاص بفرع جدة (110) من البيانات الخلفية المدمجة في الصفحة """
    try:
        # محاولة 1: البحث عن كود فرع جدة "110" متبوعاً بقيمة المخزون المتاحة له مباشرة في الـ JSON المدفون
        jeddah_json_match = re.search(r'"110"\s*:\s*\{[^}]*?"availableStock"\s*:\s*(\d+)', html_content)
        if jeddah_json_match:
            stock = int(jeddah_json_match.group(1))
            print(f"   [مخزن جدة] تم العثور على الكمية عبر النمط الأول: {stock}")
            return stock

        # محاولة 2: نمط آخر لـ buCode الخاص بفرع 110 في إيكيا
        bu_match = re.search(r'"buCode"\s*:\s*"110".*?"availableStock"\s*:\s*(\d+)', html_content, re.DOTALL)
        if bu_match:
            stock = int(bu_match.group(1))
            print(f"   [مخزن جدة] تم العثور على الكمية عبر النمط الثاني: {stock}")
            return stock

        # محاولة 3: نمط الـ quantity مع كود المتجر 110
        qty_match = re.search(r'"storeId"\s*:\s*"110".*?"quantity"\s*:\s*(\d+)', html_content, re.DOTALL)
        if qty_match:
            stock = int(qty_match.group(1))
            print(f"   [مخزن جدة] تم العثور على الكمية عبر النمط الثالث: {stock}")
            return stock

        # محاولة 4: فحص نصوص الصفحة كخيار احتياطي أخير
        text_match = re.search(r'متوفر في المخزون عدد\s*(\d+)', html_content)
        if text_match:
            return int(text_match.group(1))

    except Exception as e:
        print(f"   DEBUG: خطأ أثناء فحص مخزون جدة للمنتج {prod_id}: {e}")

    # في حال لم يتم العثور على الفرع ولكن المنتج معروض ومتوفر عاماً بالموقع نضع مخزون أمان افتراضي للطلب
    if "معرض — متوفر" in html_content or "في المخزون" in html_content:
        return 3

    print(f"   [مخزن جدة] المنتج غير متوفر حالياً (0)")
    return 0

#  FETCH IKEA DATA WITH FIXES
# ==========================================
def fetch_ikea_product(product_url, prod_id):
    try:
        headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
            "Accept-Language": "ar,en-US;q=0.9"
        }

        response = requests.get(product_url, headers=headers, timeout=30)
        if response.status_code != 200:
            print(f"❌ فشل فتح رابط إيكيا: {product_url}")
            return None

        html = response.text
        soup = BeautifulSoup(html, 'html.parser')

        # 1. الاسم
        name = None
        title_tag = soup.find('title')
        if title_tag:
            name = title_tag.get_text(strip=True).replace(" - IKEA", "").replace("IKEA", "").strip()

        # 2. السعر الأساسي (تنظيف وتحديث الـ Regex لضمان عدم الفشل)
        base_price = None
        price_match = re.search(r'"price"\s*:\s*"?(\d+)', html)
        if not price_match:
            price_match = re.search(r'SAR\s*(\d+)', html)
        if price_match:
            base_price = int(price_match.group(1))

        # 3. الوصف
        desc = ""
        desc_tag = soup.find('meta', attrs={'name': 'description'})
        if desc_tag:
            desc = desc_tag.get('content', '')

        # 4. الصور الأساسية (إصلاح روابط الـ // وترقيتها لأعلى دقة s5)
        images = []
        img_tags = soup.find_all('img')
        for img in img_tags:
            src = img.get('src') or img.get('data-src') or img.get('data-main-image')
            if src and "images/products" in src:
                # معالجة الروابط النسبية والبروتوكولات الناقصة
                if src.startswith('//'):
                    src = 'https:' + src
                elif src.startswith('/'):
                    src = 'https://www.ikea.com' + src

                # ترقية جودة الصورة تلقائياً لـ s5 (الحجم الكبير والمناسب للتطبيق)
                high_res_src = re.sub(r'_s\d\.', '_s5.', src)

                # Remove query parameters like ?f=u
                high_res_src = high_res_src.split('?')[0]

                if high_res_src not in images:
                    images.append(high_res_src)

        # احتياطي: الميتا تاج og:image
        if not images:
            og_image = soup.find('meta', property='og:image')
            if og_image and og_image.get('content'):
                src = og_image['content']
                if src.startswith('//'): src = 'https:' + src

                # Remove query parameters like ?f=u
                src = src.split('?')[0]

                images.append(src)

        # التحقق من البيانات الأساسية
        if not name or base_price is None or base_price == 0 or not images:
            return None

        # استدعاء دالة المخزون لفرع جدة 110
        quantity = extract_stock_from_html(html, prod_id)

        return {
            "name": name,
            "desc": desc,
            "base_price": base_price,
            "quantity": quantity,
            "images": images
        }

    except Exception as e:
        print("Fetch Error:", e)
        return None
